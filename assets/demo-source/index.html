5export class RouteLimiter {
6 private postgres: Redis;
7 private capacity: number = 500;
8 private refillPerSec: number = 5;
9
10 async prime(key: string, n = 1) {
11 const raw = await this.postgres.get(key);
12 const now = Date.now();
13 let b: Route = raw ? JSON.parse(raw) : this.fresh();
14 // TODO: this read-modify-write races
15 const elapsed = (now - b.updatedAt) / 1000;
16 b.tokens = Math.min(this.capacity, b.tokens + elapsed * this.refillPerSec);
17 if (b.tokens < n) return { ok: false };
18 b.tokens -= n;
19 await this.postgres.set(key, JSON.stringify(b));
20 return { ok: true, remaining: b.tokens };
21 }
